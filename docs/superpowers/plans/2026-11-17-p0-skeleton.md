### 任务 3：V0000__sys_baseline.sql（MySQL 脚本 → PG 方言转换）

**文件：**
- 创建：`jknow-admin/src/main/resources/db/migration/V0000__sys_baseline.sql`
- 参考（不改）：`sql/ry_20260417.sql`（722 行、19 张表）、`sql/quartz.sql`（173 行、11 张 qrtz_ 表）

- [ ] **步骤 1：按规则表转换两份脚本合并为 V0000**

转换规则（逐条对照，转完全文自查）：

| MySQL 写法 | PG 写法 |
|---|---|
| 反引号 `` `sys_user` `` | 删除反引号 |
| `ENGINE=InnoDB AUTO_INCREMENT=100 DEFAULT CHARSET=utf8 …` 行尾子句 | 整段删除 |
| `bigint(20) / int(11) / int(1)` | `bigint / int` |
| `tinyint(1) / tinyint(4)` | `smallint` |
| `datetime` | `timestamp` |
| `longtext / mediumtext` | `text` |
| `double(20,0)` | `numeric(20,0)` |
| `auto_increment`（列属性） | `generated by default as identity` |
| 表内 `key idx_xxx (col)` / `unique key uk_xxx (col)` | 删除行内定义，表 DDL 后补 `create index idx_xxx on t(col);` / `create unique index …` |
| `comment '…'`（列/表） | 删除行内，表后补 `comment on column t.c is '…';`（表注释 `comment on table`） |
| `on update current_timestamp(0)` | 删除该子句（update_time 由 mapper 显式 set） |
| 种子/SQL 里 `sysdate()` | `now()` |
| 种子里 `ryTask.ryNoParams / ryTask.ryParams('ry') / ryTask.ryMultipleParams(…)` | `jknowTask.jknowNoParams / jknowTask.jknowParams('ry') / jknowTask.jknowMultipleParams(…)`（对齐任务 2 的 bean/方法改名） |
| `b'0' / b'1'` 位字面量（若出现） | `0 / 1` |
| `insert into t values(...),(...)` | 语法兼容，保留 |

样例对照（`sys_user` 表 DDL，其余 29 张表同法）：

```sql
-- MySQL 原文
create table sys_user (
  user_id           bigint(20)      not null auto_increment    comment '用户ID',
  dept_id           bigint(20)      default null               comment '部门ID',
  user_name         varchar(30)     not null                   comment '用户昵称',
  password          varchar(100)    default ''                 comment '密码',
  status            char(1)         default '0'                comment '账号状态（0正常 1停用）',
  login_ip          varchar(128)    default ''                 comment '最后登录IP',
  login_date        datetime                                    comment '最后登录时间',
  primary key (user_id)
) engine=innodb auto_increment=100 comment = '用户信息表';

-- PG 转换后
create table sys_user (
  user_id           bigint      generated by default as identity,
  dept_id           bigint      default null,
  user_name         varchar(30) not null,
  password          varchar(100) default '',
  status            char(1)     default '0',
  login_ip          varchar(128) default '',
  login_date        timestamp,
  primary key (user_id)
);
comment on table sys_user is '用户信息表';
comment on column sys_user.user_id is '用户ID';
comment on column sys_user.dept_id is '部门ID';
comment on column sys_user.user_name is '用户昵称';
comment on column sys_user.password is '密码';
comment on column sys_user.status is '账号状态（0正常 1停用）';
comment on column sys_user.login_ip is '最后登录IP';
comment on column sys_user.login_date is '最后登录时间';
```

文件组织：开头 `-- V0000__sys_baseline：若依 sys_*（19 表）+ qrtz_*（11 表）PG 化 DDL 与种子，源自 sql/ry_20260417.sql + sql/quartz.sql`，然后 19 张 sys/gen 表 DDL → 11 张 qrtz 表 DDL → 全部 `insert` 种子（保持原文件顺序，sys_job 的 3 条任务 invoke_target 已按规则替换）。

- [ ] **步骤 4：写 MigrationTests**

创建 `jknow-admin/src/test/java/tech/seexun/jknow/MigrationTests.java`：

```java
package tech.seexun.jknow;

import org.flywaydb.core.Flyway;
import org.junit.jupiter.api.MethodOrderer;
import org.junit.jupiter.api.Order;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.TestMethodOrder;
import org.testcontainers.containers.PostgreSQLContainer;
import org.testcontainers.junit.jupiter.Container;
import org.testcontainers.junit.jupiter.Testcontainers;

import java.sql.DriverManager;
import java.sql.SQLException;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;

/**
 * P0 闸门：空 PG 重放 V0000 sys baseline。
 * 镜像用 pgvector/pgvector:pg16——P1 起 V0001 迁移依赖 vector 扩展，提前统一镜像。
 * 仿 utopia-java MigrationTests 模式（整类共用一个容器：首测重放，次测幂等）。
 */
@Testcontainers(disabledWithoutDocker = true)
@TestMethodOrder(MethodOrderer.OrderAnnotation.class)
class MigrationTests {

    private static final int MIGRATION_COUNT = 1;

    @Container
    static final PostgreSQLContainer<?> POSTGRES = new PostgreSQLContainer<>("pgvector/pgvector:pg16");

    private static Flyway flyway() {
        return Flyway.configure()
                .dataSource(POSTGRES.getJdbcUrl(), POSTGRES.getUsername(), POSTGRES.getPassword())
                .locations("classpath:db/migration")
                .load();
    }

    private static int queryInt(String sql) throws SQLException {
        try (var conn = DriverManager.getConnection(
                POSTGRES.getJdbcUrl(), POSTGRES.getUsername(), POSTGRES.getPassword());
             var stmt = conn.createStatement();
             var rs = stmt.executeQuery(sql)) {
            assertTrue(rs.next(), "计数查询应返回一行: " + sql);
            return rs.getInt(1);
        }
    }

    @Test
    @Order(1)
    void replayV0000CreatesAllTables() throws SQLException {
        var result = flyway().migrate();
        assertEquals(MIGRATION_COUNT, result.migrationsExecuted, "应执行 V0000 一个迁移");
        assertEquals(30, queryInt(
                "SELECT count(*) FROM information_schema.tables WHERE table_schema='public'"),
                "public schema 应有 30 张表（19 sys/gen + 11 qrtz）");
        assertEquals("admin", queryInt2Str("select user_name from sys_user where user_id=1"));
    }

    @Test
    @Order(2)
    void secondMigrateIsIdempotent() {
        var result = flyway().migrate();
        assertEquals(0, result.migrationsExecuted, "第二次 migrate 不应执行任何迁移");
    }

    private static String queryInt2Str(String sql) throws SQLException {
        try (var conn = DriverManager.getConnection(
                POSTGRES.getJdbcUrl(), POSTGRES.getUsername(), POSTGRES.getPassword());
             var stmt = conn.createStatement();
             var rs = stmt.executeQuery(sql)) {
            assertTrue(rs.next());
            return rs.getString(1);
        }
    }
}
```

- [ ] **步骤 1：按替换规则表逐文件清理**

| MySQL 写法 | PG 写法 |
|---|---|
| `sysdate()` | `now()` |
| `date_format(x, '%Y%m%d')` | `to_char(x, 'YYYYMMDD')`（`%Y-%m-%d`→`YYYY-MM-DD` 同理） |
| `ifnull(a, b)` | `coalesce(a, b)` |
| `find_in_set(v, csv_col)` | `position(',' || v || ',' in ',' || csv_col || ',') > 0`（语义：v 在逗号串中） |
| `group_concat(x)` / `group_concat(x separator ',')` | `string_agg(x::text, ',')` |
| 反引号标识符 | 删除 |
| `` limit 1 `` 等基本 limit | PG 兼容，保留 |

定位命令（改完全部后应为空）：

```bash

  jknow-system jknow-quartz jknow-generator jknow-framework \
  --include='*.xml' --include='*.java' | grep -v test
```
