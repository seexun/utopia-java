```yaml
services:
  pg:
    image: pgvector/pgvector:pg16
    environment:
      POSTGRES_PASSWORD: postgres
      POSTGRES_DB: jknow
    ports: ["15432:5432"]
    volumes: [jknow-pg-data:/var/lib/postgresql/data]
  redis:
    image: redis:7-alpine
    ports: ["36379:6379"]

    image: utopia-es-ik:8.14.3

      - xpack.security.enabled=false
      - ES_JAVA_OPTS=-Xms512m -Xmx512m
    ports: ["9200:9200", "9300:9300"]
    volumes: [jknow-es-data:/usr/share/elasticsearch/data]
volumes:
  jknow-pg-data:
  jknow-es-data:
```

```sql

  SELECT utopia_uuid INTO admin_uuid FROM sys_user WHERE user_id = 1;
```
